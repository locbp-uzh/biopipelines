# LocalColabFold on Daint (aarch64/GH200).
# The upstream install_colabbatch_linux.sh cannot be used here: it fetches
# Mambaforge-Linux-x86_64.sh and installs jax[cuda11_pip] from Google's release
# index, which publishes x86-64 wheels only. Both halves have aarch64 equivalents
# on PyPI, so the env is built directly instead.
# jax 0.10.2 reports [CudaDevice(id=0)] on a GH200. colabfold_search (what
# MMseqs2Server calls) also needs an `mmseqs` binary, which this does not supply.
jax[cuda12]
# The extra is `alphafold-minus-jax`, upstream's own spelling. pip only WARNS on
# an unknown extra, so a wrong name installs colabfold with no alphafold-colabfold
# and colabfold_batch dies at import with "alphafold is not installed".
colabfold[alphafold-minus-jax] @ git+https://github.com/sokrypton/ColabFold
